### 标签
单调栈，模板题

### 时间
$O(n)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
void solve()
{
    int n;
    cin >> n;
    vector<int>a(n + 1);
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
    }
    stack<int>s;
    vector<int>ans(n + 1);
    for (int i = n; i; i--)
    {
        while (!s.empty() && a[i] >= a[s.top()]) s.pop();
        ans[i] = (s.empty() ? 0 : s.top());
        s.push(i);
    }
    for (int i = 1; i <= n; i++)
    {
        cout << ans[i] << '\n';
    }
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
