#### 标签
$dfs$， $bfs$，递推

#### 思路
首先想到暴力枚举每个点跑 $bfs$，计算到所有点的最短距离和，但是这样时间复杂度是 $n^2$，会 $TLE$。

由于题目约束图中没有大小 $\ge3$ 的环，所以一定是一棵树。
那么可以想到类似换根 $dp$，通过只 $bfs$ 一个点求出所有点的答案 $mx_i$。
先钦定 $1$ 作为根，通过 $dfs$ 求出所有点的子树大小 $sz_i$，考虑一条边 $u-v$，递推关系是 $mx_v=mx_u-sz_v+(n-sz_v)$，意义是 $v$ 的子树所有点距离 $-1$，其他点距离 $+1$。
距离最小的距离和，按序输出答案即可。

#### 时间
$O(n)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 5e4 + 10;
int sz[N];
vector<vector<int> >g(N);
void dfs(int u, int fa)
{
    sz[u] = 1;
    for (int v : g[u])
    {
        if (v == fa) continue;
        dfs(v, u);
        sz[u] += sz[v];
    }
}
void solve()
{
    int n, m;
    cin >> n >> m;
    for (int i = 0, u, v; i < m; i++)
    {
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 0);
    vector<ll>mx(n + 1);
    vector<int>vis(n + 1), dis(n + 1);
    queue<int>q;
    q.push(1);
    vis[1] = 1;
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        for (int v : g[u])
        {
            if (!vis[v])
            {
                vis[v] = 1;
                dis[v] = dis[u] + 1;
                q.push(v);
            }
        }
    }
    ll tmp = 0;
    for (int i = 1; i <= n; i++)
    {
        tmp += dis[i];
    }
    mx[1] = tmp;
    vis.assign(n + 1, 0);
    int mn = mx[1];
    q.push(1);
    vis[1] = 1;
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        for (int v : g[u])
        {
            if (!vis[v])
            {
                vis[v] = 1;
                mx[v] = mx[u] - sz[v] + (n - sz[v]);
                if (mx[v] < mn) mn = mx[v];
                q.push(v);
            }
        }
    }
    for (int i = 1; i <= n; i++)
    {
        if (mx[i] == mn) cout << i << " ";
    }
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
