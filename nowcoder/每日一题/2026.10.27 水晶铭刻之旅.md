### 标签
强连通分量，缩点

### 思路
由于回路可以重复走，可以想到强连通分量，即一个强连通分量内选一个起点整个强连通分量都被铭刻了。
考虑缩点维护所有强连通分量的最小值，第一问答案就是这些值之和，第二问答案是所有强连通分量最小值数量之积，记得取模。

### 时间
$O(n + m)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e5 + 10;
const int mod = 1e9 + 7;
int w[N], mn[N], cnt[N];
int tot, idx, dfn[N], low[N], bel[N];
vector<vector<int> >g(N);
bool ins[N];
stack<int>s;
void tarjan(int u)
{
    dfn[u] = low[u] = ++tot;
    s.push(u);
    ins[u] = true;
    for (int v : g[u])
    {
        if (!dfn[v])
        {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        }
        else if (ins[v])
        {
            low[u] = min(low[u], dfn[v]);
        }
    }
    if (low[u] == dfn[u])
    {
        idx++;
        int x;
        do
        {
            x = s.top();
            s.pop();
            ins[x] = false;
            bel[x] = idx;
            mn[idx] = min(mn[idx], w[x]);
        } while (x != u);
    }
}
void solve()
{
    int n, m;
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        cin >> w[i];
        mn[i] = 1e9;
    }
    cin >> m;
    for (int i = 0, u, v; i < m; i++)
    {
        cin >> u >> v;
        g[u].push_back(v);
    }
    for (int i = 1; i <= n; i++)
    {
        if (!dfn[i]) tarjan(i);
    }
    ll ans1 = 0, ans2 = 1;
    for (int i = 1; i <= n; i++)
    {
        if (mn[bel[i]] == w[i]) cnt[bel[i]]++;
    }
    for (int i = 1; i <= idx; i++)
    {
        ans1 += mn[i];
        ans2 = ans2 * cnt[i] % mod;
    }
    cout << ans1 << " " << ans2;
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
