### 标签
二分答案

### 思路
答案具有单调性，直接二分。

当前值大于答案的部分需要用 $x$ 弥补，操作次数不超过 $k$ 这个答案是合理的，收拢右边界。

注意答案可能会爆 $int$。

### 时间
$O(nlogn)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e5 + 10;
int n;
ll k, x, a[N];
bool check(ll mid)
{
    ll res = 0;
    for (int i = 1; i <= n; i++)
    {
        if (a[i] > mid)
        {
            ll d = a[i] - mid;
            res += (d + x - 1) / x;
        }
        if (res > k) return false;
    }
    return (res <= k);
}
void solve()
{
    cin >> n >> k >> x;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
    }
    ll l = -1e18, r = 1e18, ans = 1e18;
    while (l <= r)
    {
        ll mid = (l + r) >> 1;
        if (check(mid))
        {
            ans = min(ans, mid);
            r = mid - 1;
        }
        else l = mid + 1;
    }
    cout << ans;
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
