### 标签
枚举

### 思路
由于 $n<=5000$，可以考虑 $n^2$ 枚举从每个点出发的最长一段。

每一天声望只加一次，但耐久会随打怪消耗，所以最优是每天只打一只怪。

枚举时同时检查声望和耐久会不会超过阈值即可。

### 时间
$O(n)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
void solve()
{
    int n;
    ll m, u;
    cin >> n >> m >> u;
    vector<ll>a(n + 1), b(n + 1);
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 1; i <= n; i++) cin >> b[i];
    int ans = 0;
    for (int i = 1; i <= n; i++)
    {
        ll c1 = 0, c2 = 0;
        for (int j = i; j <= n; j++)
        {
            c1 += a[j];
            c2 += b[j];
            if (c1 > m || c2 > u) break;
            ans = max(ans, j - i + 1);
        }
    }
    cout << ans;
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
