### 思路
题目的操作相当于一个格子周围有 $x$ 个格子，这个格子中的数会贡献答案 $x$ 次，不难发现矩阵中的相邻格子数最多只有 $2,3,4$ 这三种情况。

要最大化权值，贪心地摆放，把值大的摆在 $4$ 处，再是 $3$ 处。于是 $1\sim n^2$ 上所有的数切成三个部分 $$1,\cdots,4,5,\cdots,4n-4,4n-3,\cdots,n^2$$
各自的项数为 $4,4n-8,(n-2)^2$，用等差数列求和公式计算各项并乘以系数即可。

### 时间
$O(1)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 1e9 + 7;
void solve()
{
    ll n;
    cin >> n;
    ll ans = 20;
    ll tot1 = n * n % mod;
    ll tot2 = (n - 2) * (n - 2) % mod;
    if (n > 2)
    {
        ans = (ans + (4 * tot1 - 7 * n % mod - 2 + mod) * 6 % mod) % mod; 
        ans = (ans + (4 * n % mod + tot1 - 3 + mod) * tot2 % mod * 2 % mod) % mod;
    }
    cout << ans;
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
