### 标签
$KMP$

### 思路

要求不重叠的前后缀数量，也就是 $next[i]<=\lfloor\frac{i}{2}\rfloor$。$next$ 数组有性质，前缀 $[1,i]$ 的所有公共前后缀长度，恰好是 $next[i],next[next[i]]...$，直到长度为 $0$。

在 $KMP$ 求 $next$ 时，可以顺带求出 $[1,i]$ 中的公共前后缀数量，即 $num[i]=num[next[i]]+1$，注意此时包含了 $[1,i]$ 这个串， $num[1]=1$。

再次匹配，对于每个 $i$，跳到 $2next[i]\le i$ 处，此时 $num[next[i]]$ 就是这个 $i$ 的答案。

### 时间
单组: $O(L)$ 其中 $L$ 为字符串长度

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e6 + 10;
const int mod = 1e9 + 7;
void solve()
{
    string s;
    cin >> s;
    int n = s.length();
    s = " " + s;
    vector<int>nxt(n + 1);
    vector<ll>num(n + 1);
    num[1] = 1;
    for (int i = 2, j = 0; i <= n; i++)
    {
        while (j && s[i] != s[j+1]) j = nxt[j];
        if (s[i] == s[j+1]) j++;
        nxt[i] = j;
        num[i] = num[j] + 1;
    }
    ll ans = 1;
    for (int i = 2, j = 0; i <= n; i++)
    {
        while (j && s[i] != s[j+1]) j = nxt[j];
        if (s[i] == s[j+1]) j++;
        while ((j << 1) > i) j = nxt[j];
        ans = ans * (num[j] + 1) % mod;
    }
    cout << ans << '\n';
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
