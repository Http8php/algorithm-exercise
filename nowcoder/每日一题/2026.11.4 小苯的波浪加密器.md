### 思路
$xy\%10=((x\%10)\times (y\%10))\%10$，所以乘积的个位数只与两个数的个位有关。

枚举所有的个位数进行组合，检查 $a_1a_2\%10=a_3$ 和 $a_2a_3\%10=a_4$。枚举两个区间的前 $10$ 个数，既满足所有个位数组合的要求，又满足最小字典序的要求。

注意 $n=3$ 时没有 $a_4$，需要特判。

### 时间
$O(n + 100)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
void solve()
{
    int n;
    ll l1, r1, l2, r2;
    cin >> n >> l1 >> r1 >> l2 >> r2;
    ll a3, a4;
    cin >> a3;
    if (n > 3) cin >> a4;
    for (int i = 0, x; i < n - 4; i++)
    {
        cin >> x;
    }
    for (ll a1 = l1; a1 <= min(r1, l1 + 9); a1++)
    {
        for (ll a2 = l2; a2 <= min(r2, l2 + 9); a2++)
        {
            if (a1 * a2 % 10 == a3)
            {
                if (n == 3)
                {
                    cout << a1 << " " << a2 << '\n';
                    return;
                }
                else if (a2 * a3 % 10 == a4)
                {
                    cout << a1 << " " << a2 << '\n';
                    return;
                }
            }
        }
    }
    cout << -1 << " " << -1 << '\n';
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
