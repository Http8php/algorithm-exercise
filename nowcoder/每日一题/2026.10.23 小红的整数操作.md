#### 思路
发现两种操作不改变 $\frac{x}{y}$，于是可以先约到最简分数后再成倍扩大。
设 $x\le y$，可行的左端点 $L$ 是 $\lceil\frac{l}{x}\rceil$，右端点 $R$ 是 $\lfloor\frac{r}{y}\rfloor$
答案就是 $\max(0, R - L + 1)$

#### 时间
$O(logmin(x, y))$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
void solve()
{
    ll x, y, l, r;
    cin >> x >> y >> l >> r;
    while (gcd(x, y) > 1)
    {
        ll g = gcd(x, y);
        x /= g;
        y /= g;
    }
    ll L = (l + min(x, y) - 1) / min(x, y), R = r / max(x, y);
    ll ans = max(0ll, R - L + 1);
    cout << ans;
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
