#### 标签
前缀和， $ST$ 表

#### 思路
发现 $[l-1,r]$ 中的前缀和最大值 $-$ 最小值就是答案。
处理出前缀和后用 $ST$ 表查询即可。

#### 时间
$O(nlogn + q)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 5e5 + 10;
int n, q, l2[N];
ll a[N], mx[N][25], mn[N][25];
void init()
{
    for (int i = 2; i <= n + 1; i++)
    {
        l2[i] = l2[i/2] + 1;
    }
    for (int i = 1; i <= n; i++)
    {
        mx[i][0] = mn[i][0] = a[i];
    }
    for (int i = 1; (1 << i) <= n + 1; i++)
    {
        for (int j = 0; j + (1 << i) - 1 <= n; j++)
        {
            mx[j][i] = max(mx[j][i-1], mx[j+(1<<(i-1))][i-1]);
            mn[j][i] = min(mn[j][i-1], mn[j+(1<<(i-1))][i-1]);
        }
    }
}
ll query_max(int l, int r)
{
    int j = l2[r-l+1];
    return max(mx[l][j], mx[r-(1<<j)+1][j]);
}
ll query_min(int l, int r)
{
    int j = l2[r-l+1];
    return min(mn[l][j], mn[r-(1<<j)+1][j]);
}
void solve()
{
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
        a[i] += a[i-1];
    }
    init();
    cin >> q;
    while (q--)
    {
        int l, r;
        cin >> l >> r;
        cout << query_max(l - 1, r) - query_min(l - 1, r) << '\n';
    }
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
