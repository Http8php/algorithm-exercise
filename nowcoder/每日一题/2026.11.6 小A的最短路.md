### 标签
$LCA$

### 思路
给出的图是一棵树，且边权是 $1$，可以用 $LCA$ 求最短路，设 $dep[i]$ 为 $i$ 点的深度， $fa$ 为 $a,b$ 两点的最近公共祖先，则 $a,b$ 两点的最短路距离 $dis(a,b)$ 为 $dep[a] + dep[b] - 2dep[fa]$。

现在还有一条边权为 $0$ 的边 $u-v$，需要额外考虑经过这条边的情况，即 $dis(x,u)+dis(v,y)$ 和 $dis(x,v)+dis(u,y)$，与 $dis(x,y)$ 取最小值即可。

### 时间
$O(n + qlogn)$

$Fun Fact:$ 我一直以为多维数组把小维度放前面一定可以减小常数，但是这题写 $f[20][N]$ 一直 $TLE$，原来对于频繁按行访问的二维数组，应将最常变动的维度放在最后，求 $LCA$ 中的循环频繁访问小维度 $i$ ，应该放在后面使缓存连续。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 3e5 + 10;
vector<vector<int> >g(N);
int dep[N], f[N][20];
void dfs(int u, int fa)
{
    dep[u] = dep[fa] + 1;
    f[u][0] = fa;
    for (int i = 1; i <= 19; i++)
    {
        f[u][i] = f[f[u][i-1]][i-1];
    }
    for (int v : g[u])
    {
        if (v == fa) continue;
        dfs(v, u);
    }
}
int lca(int a, int b)
{
    if (dep[a] < dep[b]) swap(a, b);
    for (int i = 19; i >= 0; i--)
    {
        if (dep[f[a][i]] >= dep[b])
        {
            a = f[a][i];
        }
    }
    if (a == b) return a;
    for (int i = 19; i >= 0; i--)
    {
        if (f[a][i] != f[b][i])
        {
            a = f[a][i];
            b = f[b][i];
        }
    }
    return f[a][0];
}
ll calc(int a, int b)
{
    int fa = lca(a, b);
    return dep[a] - dep[fa] + dep[b] - dep[fa];
}
void solve()
{
    int n, a, b, u, v, q;
    cin >> n;
    for (int i = 0; i < n - 1; i++)
    {
        cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1, 0);
    cin >> a >> b >> q;
    while (q--)
    {
        cin >> u >> v;
        ll ans = min({calc(u, v), calc(u, a) + calc(v, b), calc(u, b) + calc(v, a)});
        cout << ans << '\n';
    }
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
