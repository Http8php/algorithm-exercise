### 标签
线段树

### 思路
题目要求多次询问区间，考虑线段树。操作 $1$ 求和是线段树正常操作，但是操作 $2$ 区间异或，由于区间异或与 $a_i$ 的值有关，所以不能简单地用懒标记处理。

异或 $k$ 会翻转每位 $k$ 的二进制位为 $1$ 的二进制位，对每个 $bit$ 建立一棵线段树 $tr[i][N]$，如果 $a_i$ 在该位为 $1$，线段树的叶子节点就是 $1$。

此时问题变成经典的区间异或 $1$，如果某个长度为 $len$ 的 $p$ 节点要翻转，则 $tr[i][p]=len-tr[i][p]$。此时求区间和就是区间内 $1$ 的数量 $cnt$，由于 $a_i\le10^5$，所以最多 $17$ 个二进制位，答案就是 $\sum_{i=0}^{16}cnt\times2^i$。

### 时间
$O(n + mlogn)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e5 + 10;
int n, m;
ll a[N], b[N], tr[20][N<<2], mark[20][N<<2];
void push_down(int p, ll len, int i)
{
    if (len <= 1 || mark[i][p] == 0) return;
    tr[i][p<<1] = (len - len / 2) - tr[i][p<<1];
    mark[i][p<<1] ^= 1;
    tr[i][p<<1|1] = len / 2 - tr[i][p<<1|1];
    mark[i][p<<1|1] ^= 1;
    mark[i][p] = 0;
}
void build(ll l, ll r, int i, int p = 1)
{
    if (l == r)
    {
        tr[i][p] = b[l];
        return;
    }
    ll mid = (l + r) >> 1;
    build(l, mid, i, p << 1);
    build(mid + 1, r, i, p << 1 | 1);
    tr[i][p] = tr[i][p<<1] + tr[i][p<<1|1];
}
ll query(ll l, ll r, int i, int p = 1, ll cl = 1, ll cr = n)
{
    if (cl >= l && cr <= r) return tr[i][p];
    push_down(p, cr - cl + 1, i);
    ll mid = (cl + cr) >> 1, res = 0;
    if (mid >= l) res += query(l, r, i, p << 1, cl, mid);
    if (mid < r) res += query(l, r, i, p << 1 | 1, mid + 1, cr);
    return res;
}
void update(ll l, ll r, int i, int p = 1, ll cl = 1, ll cr = n)
{
    if (cl >= l && cr <= r)
    {
        tr[i][p] = (cr - cl + 1) - tr[i][p];
        mark[i][p] ^= 1;
        return;
    }
    push_down(p, cr - cl + 1, i);
    ll mid = (cl + cr) >> 1;
    if (mid >= l) update(l, r, i, p << 1, cl, mid);
    if (mid < r) update(l, r, i, p << 1 | 1, mid + 1, cr);
    tr[i][p] = tr[i][p<<1] + tr[i][p<<1|1];
}
void solve()
{
    cin >> n >> m;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
    }
    for (int j = 0; j < 17; j++)
    {
        for (int i = 1; i <= n; i++)
        {
            if ((a[i] >> j) & 1) b[i] = 1;
            else b[i] = 0;
        }
        build(1, n, j);
    }
    while (m--)
    {
        int op, l, r;
        cin >> op >> l >> r;
        if (op == 1)
        {
            ll ans = 0;
            for (int i = 0; i < 17; i++)
            {
                ans += query(l, r, i) * (1ll << i);
            }
            cout << ans << '\n';
        }
        else
        {
            int k;
            cin >> k;
            for (int i = 0; i < 17; i++)
            {
                if ((k >> i) & 1)
                {
                    update(l, r, i);
                }
            }
        }
    }
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
