### 标签
线段树，模板题

### 思路
区间加和单点查询是线段树/树状数组的基础用法，难点在于怎样维护区间乘。

对于一个数 $x$，加法为 $1x+b$，乘法为 $ax+0$，假设进行变换的数为 $ax+b$，现有新乘数 $A$，于是 $A(ax+b)=Aax+Ab$，可以发现乘法作用于乘法和加法上。

维护两个懒标记 $add,mul$，遇到在整个范围内的区间乘 $k$ 就把这个线段 $\times k$，下传标记时，由于 $add$ 已经被乘过了，子节点更新时乘父节点的 $mul$ 直接加 $add$， $add$ 更新时先把子节点的懒标记乘父节点的 $mul$ 再加父节点的 $add$。

注意 $x,k$ 有负数情况，取模时先 $+mod$ 再 $\\%mod$。

### 时间
$O(n + qlogn)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e5 + 10;
const int mod = 998244353;
int n, q;
ll a[N], tr[N<<2], add[N<<2], mul[N<<2];
void build(int p, int l, int r)
{
    mul[p] = 1;
    if (l == r)
    {
        tr[p] = (a[l] + mod) % mod;
        return;
    }
    int mid = (l + r) >> 1;
    build(p << 1, l, mid);
    build(p << 1 | 1, mid + 1, r);
    tr[p] = (tr[p<<1] + tr[p<<1|1]) % mod;
}
void push_down(int p, ll len)
{
    if (len <= 1) return;
    tr[p<<1] = ((tr[p<<1] * mul[p] % mod + add[p] * (len - len / 2) % mod) % mod + mod) % mod;
    mul[p<<1] = (mul[p<<1] * mul[p] % mod + mod) % mod;
    add[p<<1] = ((add[p<<1] * mul[p] % mod + add[p]) % mod + mod) % mod;
    tr[p<<1|1] = ((tr[p<<1|1] * mul[p] % mod + add[p] * (len / 2) % mod) % mod + mod) % mod;
    mul[p<<1|1] = (mul[p<<1|1] * mul[p] % mod + mod) % mod;
    add[p<<1|1] = ((add[p<<1|1] * mul[p] % mod + add[p]) % mod + mod) % mod;
    add[p] = 0, mul[p] = 1;
}
void update(int l, int r, ll d, int i, int p = 1, int cl = 1, int cr = n)
{
    if (cl >= l && cr <= r)
    {
        if (i == 1)
        {
            tr[p] = ((tr[p] + d * (cr - cl + 1) % mod) % mod + mod) % mod;
            add[p] = ((add[p] + d) % mod + mod) % mod;
        }
        else
        {
            tr[p] = (tr[p] * d % mod + mod) % mod;
            mul[p] = (mul[p] * d % mod + mod) % mod;
            add[p] = (add[p] * d % mod + mod) % mod;
        }
        return;
    }
    push_down(p, cr - cl + 1);
    int mid = (cl + cr) >> 1;
    if (mid >= l) update(l, r, d, i, p << 1, cl, mid);
    if (mid < r) update(l, r, d, i, p << 1 | 1, mid + 1, cr);
    tr[p] = ((tr[p<<1] + tr[p<<1|1]) % mod + mod) % mod;
}
ll query(int x, int p = 1, int cl = 1, int cr = n)
{
    if (cl == x && cr == x) return tr[p];
    push_down(p, cr - cl + 1);
    int mid = (cl + cr) >> 1;
    ll res = 0;
    if (mid >= x) res += query(x, p << 1, cl, mid);
    if (mid < x) res += query(x, p << 1 | 1, mid + 1, cr);
    return (res % mod + mod) % mod;
}
void solve()
{
    cin >> n >> q;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
    }
    build(1, 1, n);
    while (q--)
    {
        int op, l, r;
        ll x;
        cin >> op;
        if (op == 1 || op == 2)
        {
            cin >> l >> r >> x;
            update(l, r, x, op);
        }
        else
        {
            cin >> x;
            cout << query(x) << '\n';
        }
    }
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
