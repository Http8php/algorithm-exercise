### 标签
动态规划

### 思路
设 $dp[i][j]$ 为前 $i$ 个数余数为 $j$ 的集合个数。

由于空集也算方案，对于所有 $i$， $dp[i][0]=1$。

转移时先算出当前数 $x$ 可以组成余数为 $0,1,2$ 时的方案数 $s_0,s_1,s_2$，可以分类 $x\%3$ 的余数讨论，比如 $x=4,x\%3=1,cnt_4=4$，此时 $s_0$ 只有 $4,4,4$ 这一种情况， $s_0=\lfloor\frac{cnt_4}{3}\rfloor$， $s_1$ 有 $\set{\set{4},\set{4,4,4,4}}$ 这两种情况， $s_1=\lceil\frac{cnt_4}{3}\rceil$，其他情况同理。

转移方程为 $dp[i][0]=dp[i-1][0]+dp[i-1][0]\times s0+dp[i-1][1]\times s2+dp[i-1][2]\times s1$， $dp[i][1]$ 和 $dp[i][2]$ 转移同理，固定转移 $9$ 次即可求出答案 $dp[n][0]$。

### 时间
$O(1)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int mod = 1e9 + 7;
void solve()
{
    vector<ll>cnt(10);
    vector<vector<ll> >dp(10, vector<ll>(3));
    for (int i = 1; i <= 9; i++)
    {
        cin >> cnt[i];
        dp[i][0] = 1;
    }
    for (int i = 1; i <= 9; i++)
    {
        ll s0, s1, s2;
        if (i % 3 == 1)
        {
            s1 = (cnt[i] + 2) / 3;
            s2 = (cnt[i] + 1) / 3;
            s0 = cnt[i] / 3;
        }
        else if (i % 3 == 2)
        {
            s1 = (cnt[i] + 1) / 3;
            s2 = (cnt[i] + 2) / 3;
            s0 = cnt[i] / 3;
        }
        else s0 = cnt[i], s1 = 0, s2 = 0;
        dp[i][0] = (dp[i-1][0] + dp[i-1][0] * s0 % mod + dp[i-1][1] * s2 % mod + dp[i-1][2] * s1 % mod) % mod;
        dp[i][1] = (dp[i-1][1] + dp[i-1][0] * s1 % mod + dp[i-1][1] * s0 % mod + dp[i-1][2] * s2 % mod) % mod;
        dp[i][2] = (dp[i-1][2] + dp[i-1][0] * s2 % mod + dp[i-1][1] * s1 % mod + dp[i-1][2] * s0 % mod) % mod;
    }
    cout << dp[9][0];
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
