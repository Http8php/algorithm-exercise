### 标签
二分，离散化，树状数组

### 思路
如果只要记录 $a_ia_j=k$ 的情况，开一个 $map$，从左往右遍历数组，如果 $k\%a_i=0$，就把 $map[\frac{k}{a_i}]$ 加入答案，随后把 $a_i$ 加入 $map$。

现在还需要统计 $>k,<k$ 的情况，考虑把 $map$ 换成树状数组，分 $a_i$ 是否能整除 $k$ 进行讨论，能整除设 $tar=\frac{k}{a_i}$，去数组中二分找数， $>tar$ 的所有数归到第一类， $=tar$ 第二类， $<tar$ 的所有数第三类；不能整除令 $tar=\lceil \frac{k}{a_i}\rceil$，后面同理讨论。注意处理 $tar$ 比所有数都大的情况。

由于 $a_i$ 值域较大，需要离散化。

### 时间
$O(nlogn)$

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
struct Fenwick
{
    int n;
    vector<ll>tr;
    Fenwick(int n): n(n), tr(n + 1) {}
    void add(int x, int v)
    {
        while (x <= n)
        {
            tr[x] += v;
            x += (x & -x);
        }
    }
    ll sum(int x)
    {
        ll ans = 0;
        while (x > 0)
        {
            ans += tr[x];
            x -= (x & -x);
        }
        return ans;
    }
};
void solve()
{
    int n;
    ll k;
    cin >> n >> k;
    vector<ll>a(n + 1), b(n + 1);
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
    }
    b = a;
    sort(b.begin() + 1, b.end());
    auto it = unique(b.begin() + 1, b.end());
    vector<int>idx(n + 1);
    for (int i = 1; i <= n; i++)
    {
        idx[i] = lower_bound(b.begin() + 1, it, a[i]) - b.begin();
    }
    Fenwick cnt(n);
    ll ans1 = 0, ans2 = 0, ans3 = 0;
    for (int i = 1; i <= n; i++)
    {
        if (k % a[i] == 0)
        {
            ll tar = k / a[i];
            int j = lower_bound(b.begin() + 1, it, tar) - b.begin();
            if (j <= n)
            {
                if (b[j] == tar)
                {
                    ans1 += cnt.sum(n) - cnt.sum(j);
                    ans2 += cnt.sum(j) - cnt.sum(j - 1);
                    ans3 += cnt.sum(j - 1);
                }
                else
                {
                    ans1 += cnt.sum(n) - cnt.sum(j - 1);
                    ans3 += cnt.sum(j - 1);
                }
            }
            else
            {
                ans3 += cnt.sum(n);
            }
        }
        else
        {
            ll tar = (k + a[i] - 1) / a[i];
            int j = lower_bound(b.begin() + 1, it, tar) - b.begin();
            if (j <= n) ans1 += cnt.sum(n) - cnt.sum(j - 1);
            ans3 += cnt.sum(j - 1);
        }
        cnt.add(idx[i], 1);
    }
    cout << ans1 << " " << ans2 << " " << ans3;
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t = 1;
    //cin >> t;
    while (t--)
    {
        solve();
    }
    return 0;
}
```
